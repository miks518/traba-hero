    ```typescript
    import type { ScanResult, MatchResult, ApiErrorResponse } from '../types';

    const API_BASE = 'http://localhost:8000';

    class ApiRequestError extends Error {
      status: number;
      details?: string;

      constructor(status: number, message: string, details?: string) {
        super(message);
        this.name = 'ApiRequestError';
        this.status = status;
        this.details = details;
      }
    }

    async function request<T>(
      endpoint: string,
      options: RequestInit = {},
      timeoutMs = 15000
    ): Promise<T> {
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), timeoutMs);

      try {
        const response = await fetch(`${API_BASE}${endpoint}`, {
          signal: controller.signal,
          ...options,
        });

        if (!response.ok) {
          let errorBody: ApiErrorResponse | null = null;
          try {
            errorBody = await response.json();
          } catch {}
          throw new ApiRequestError(
            response.status,
            errorBody?.message ?? `HTTP ${response.status}: ${response.statusText}`,
            errorBody?.details
          );
        }

        return response.json() as Promise<T>;
      } catch (err) {
        if (err instanceof ApiRequestError) throw err;
        if ((err as Error).name === 'AbortError') {
          throw new ApiRequestError(0, 'Request timed out');
        }
        throw new ApiRequestError(0, (err as Error).message);
      } finally {
        clearTimeout(timeout);
      }
    }

    function dataUrlToBlob(dataUrl: string): Blob {
      const parts = dataUrl.split(',');
      const mimeMatch = parts[0].match(/:(.*?);/);
      const mime = mimeMatch ? mimeMatch[1] : 'image/png';
      const bytes = atob(parts[1]);
      const arr = new Uint8Array(bytes.length);
      for (let i = 0; i < bytes.length; i++) {
        arr[i] = bytes.charCodeAt(i);
      }
      return new Blob([arr], { type: mime });
    }

    /**
     * Sends a cropped screenshot (as data URL) to the backend /scan endpoint.
     * The image is sent as multipart/form-data under the field name "screenshot".
     */
    export async function scanScreenshot(
      imageDataUrl: string
    ): Promise<ScanResult> {
      const blob = dataUrlToBlob(imageDataUrl);
      const formData = new FormData();
      formData.append('screenshot', blob, `scan_${Date.now()}.png`);

      return request<ScanResult>('/scan', {
        method: 'POST',
        body: formData,
      });
    }

    /**
     * Sends resume match request to the backend /match endpoint.
     * The body format aligns with the existing MatchResult type shape.
     */
    export async function matchResume(
      resumeData: { filename: string; content?: string }
    ): Promise<MatchResult> {
      return request<MatchResult>('/match', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(resumeData),
      });
    }

    export { ApiRequestError };
    ```
